# Die App selbst ist hinter einer Anmeldung; es gibt nichts zu
# indizieren. Der Demo-Modus (index.html?demo) zeigt erfundene Daten

# Aufgaben waere das Gegenteil von hilfreich.

# Das ist eine Bitte und keine Sperre: robots.txt haelt niemanden ab, der

# nur im Browser. demo-daten.js setzt zusaetzlich ein noindex-Element.

User-agent: *

Disallow: /?demo
Disallow: /demo-daten.js
